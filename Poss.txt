typedef struct {
    int row;
    int col;
} Shot;

// Add a valid ship placement to the heat map.
void score_placement(
    char shots[SIZE][SIZE],
    int heat[SIZE][SIZE],
    int row,
    int col,
    int length,
    int horizontal
) {
    int hits = 0;

    // First check whether the placement is possible.
    for (int i = 0; i < length; i++) {
        int r = row + (horizontal ? 0 : i);
        int c = col + (horizontal ? i : 0);

        if (r >= SIZE || c >= SIZE) {
            return;
        }

        if (shots[r][c] == 'O' ||
            shots[r][c] == 'S') {
            return;
        }

        if (shots[r][c] == 'X') {
            hits++;
        }
    }

    // Give extra weight to placements containing hits.
    int points = 1 + hits * HIT_BONUS;

    for (int i = 0; i < length; i++) {
        int r = row + (horizontal ? 0 : i);
        int c = col + (horizontal ? i : 0);

        // Only unknown squares can be targeted.
        if (shots[r][c] == '~') {
            heat[r][c] += points;
        }
    }
}

// Recalculate the entire heat map.
void calculate_heat_map(
    char shots[SIZE][SIZE],
    int heat[SIZE][SIZE],
    int ship_sizes[],
    int ship_count
) {
    // Reset the heat map.
    for (int r = 0; r < SIZE; r++) {
        for (int c = 0; c < SIZE; c++) {
            heat[r][c] = 0;
        }
    }

    // Evaluate each remaining ship.
    for (int s = 0; s < ship_count; s++) {
        int length = ship_sizes[s];

        for (int r = 0; r < SIZE; r++) {
            for (int c = 0; c < SIZE; c++) {

                // Horizontal placement
                score_placement(
                    shots, heat, r, c, length, 1
                );

                // Vertical placement
                score_placement(
                    shots, heat, r, c, length, 0
                );
            }
        }
    }
}

// Choose randomly among the highest-scoring squares.
Shot find_best_shot(
    char shots[SIZE][SIZE],
    int heat[SIZE][SIZE]
) {
    Shot best = {-1, -1};
    int highest = -1;
    int ties = 0;

    for (int r = 0; r < SIZE; r++) {
        for (int c = 0; c < SIZE; c++) {

            if (shots[r][c] != '~') {
                continue;
            }

            int score = heat[r][c];

            if (score > highest) {
                highest = score;
                best.row = r;
                best.col = c;
                ties = 1;
            }
            else if (score == highest) {
                ties++;

                // Reservoir sampling for random ties.
                if (rand() % ties == 0) {
                    best.row = r;
                    best.col = c;
                }
            }
        }
    }

    return best;
}
void print_heat_map(int heat[SIZE][SIZE]) {
    printf("\n              HEAT MAP\n\n");

    printf("      ");
    for (int c = 1; c <= SIZE; c++) {
        printf("%5d", c);
    }
    printf("\n");

    for (int r = 0; r < SIZE; r++) {
        printf("  %c   ", 'A' + r);

        for (int c = 0; c < SIZE; c++) {
            printf("%5d", heat[r][c]);
        }

        printf("\n");
    }
}
